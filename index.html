🚀 إعداد Apps Script (مرة واحدة فقط)

  1. افتح aistudio.google.com واحصل على Gemini API Key مجاناً
  2. افتح script.google.com وأنشئ مشروعاً جديداً
  3. الصق الكود التالي واستبدل YOUR_GEMINI_KEY بمفتاحك:
const GEMINI_KEY = "YOUR_GEMINI_KEY";

function doGet(e) {
  try {
    // حفظ في Sheets
    if (e.parameter.action === "save") {
      var sheet = SpreadsheetApp.openById(e.parameter.sheetId).getActiveSheet();
      sheet.appendRow([
        new Date(),
        decodeURIComponent(e.parameter.tone        || ""),
        decodeURIComponent(e.parameter.originalText|| ""),
        decodeURIComponent(e.parameter.hook        || ""),
        decodeURIComponent(e.parameter.rewrite     || ""),
        decodeURIComponent(e.parameter.imagePrompt || "")
      ]);
      return respond({ status: "saved" });
    }

    // توليد المحتوى عبر Gemini
    var prompt = decodeURIComponent(e.parameter.prompt || "");
    var url = "https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=" + GEMINI_KEY;
    var res = UrlFetchApp.fetch(url, {
      method: "post",
      contentType: "application/json",
      payload: JSON.stringify({
        contents: [{ parts: [{ text: prompt }] }],
        generationConfig: { temperature: 0.8, maxOutputTokens: 1500 }
      }),
      muteHttpExceptions: true
    });
    var data = JSON.parse(res.getContentText());
    if (!data.candidates || !data.candidates[0]) {
      return respond({ error: "Gemini error: " + JSON.stringify(data) });
    }
    var text = data.candidates[0].content.parts[0].text;
    return respond({ text: text });

  } catch(err) {
    return respond({ error: err.toString() });
  }
}

function respond(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
